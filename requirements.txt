# Known good on Python 3.14. Versions are the ones this corpus was built with;
# earlier releases of each probably work and have not been tested.

# Renders the first page of a letter. The thumbnails are WebP, which Pillow
# writes and which is 21% smaller than JPEG at the same quality -- the margin
# that lets the whole corpus fit inside the GitHub Pages ceiling.
pypdfium2 >= 5.13.0
pillow    >= 12.3.0

# Reads the date a PDF states about itself, and its text layer. About half
# these letters carry extractable text; see issue #1.
pypdf     >= 6.16.2

# Only for build/upload_item.py, which pushes the payload to an archive.org
# item. Not needed to build or serve the explorer: the per-member data and the
# thumbnails are both served from this repository at the moment.
internetarchive >= 5.11.0
